## CREAZIONE TABELLE ##
create table if not exists "camel"."CLIENTI"(
"CODICE_CLIENTE" INT generated by default as identity primary key,
"NOME" VARCHAR(50) not null,
"COGNOME" VARCHAR(50) not null,
"DATA_DI_NASCITA" DATE not null,
"CF" VARCHAR(16) not NULL
);

CREATE TABLE if not exists "camel"."PRATICHE" (
"CODICE_PRATICA" int4 generated by default as identity primary key,
"IMPORTO" NUMERIC(12,2) NOT NULL,
"DESCRIZIONE" varchar(1024) NULL,
"CODICE_CONTRAENTE" int4 NOT null references "camel"."CLIENTI"("CODICE_CLIENTE"),
"DATA_SOTTOSCRIZIONE" DATE not null,
"DATA_CANCELLAZIONE" DATE,
"STATO_PRATICA" VARCHAR(20)
);

create table if not exists "camel"."ESITI_ELABORAZIONI"(
"CODICE_ESITO" int generated by default as identity primary key,
"CODICE_PRATICA" INT references "camel"."PRATICHE"("CODICE_PRATICA"),
"ESITO" VARCHAR(2) not null,
"MESSAGGIO_ERRORE" VARCHAR(1024),
"NOME_FILE" VARCHAR(1024),
"DATA_ELABORAZIONE" TIMESTAMP not null default now()
);
